fetch("https://graphql.anilist.co", {
    method: "POST",
    headers: {
        "Content-Type": "application/json",
        "Accept": "application/json"
    },
    body: JSON.stringify({
        query: `
        {
          Media(search: "One Piece", type: ANIME) {
            coverImage {
              extraLarge
            }
          }
        }`
    })
}).then(r => r.text()).then(console.log)
